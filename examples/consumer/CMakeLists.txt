cmake_minimum_required(VERSION 3.25)
project(LcarsConsumer LANGUAGES CXX)
find_package(LcarsUI 0.2 REQUIRED COMPONENTS Core AppearanceMaterial OPTIONAL_COMPONENTS Desktop)
add_executable(profile_alias main.cpp)
target_compile_definitions(profile_alias PRIVATE LCARS_PACKAGE_VERSION=u"${LcarsUI_VERSION}")
target_link_libraries(profile_alias PRIVATE LcarsUI::Core LcarsUI::AppearanceMaterial)
enable_testing()
add_test(NAME profile_alias COMMAND profile_alias)

if(LcarsUI_Desktop_FOUND)
  add_executable(notices_smoke notices_smoke.cpp)
  target_compile_definitions(notices_smoke PRIVATE LCARS_THIRD_PARTY_NOTICES_FILE="${LcarsUI_THIRD_PARTY_NOTICES_FILE}")
  target_link_libraries(notices_smoke PRIVATE LcarsUI::Desktop)
  add_test(NAME notices_smoke COMMAND notices_smoke)
endif()
