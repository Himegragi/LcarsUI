cmake_minimum_required(VERSION 3.25)
project(IsoduoConsumer LANGUAGES CXX)
find_package(Isoduo 0.4 REQUIRED COMPONENTS Core AppearanceMaterial OPTIONAL_COMPONENTS Desktop)
add_executable(profile_alias main.cpp)
target_compile_definitions(profile_alias PRIVATE ISODUO_PACKAGE_VERSION=u"${Isoduo_VERSION}")
target_link_libraries(profile_alias PRIVATE Isoduo::Core Isoduo::AppearanceMaterial)
enable_testing()
add_test(NAME profile_alias COMMAND profile_alias)

if(Isoduo_Desktop_FOUND)
  add_executable(notices_smoke notices_smoke.cpp)
  target_compile_definitions(notices_smoke PRIVATE ISODUO_THIRD_PARTY_NOTICES_FILE="${Isoduo_THIRD_PARTY_NOTICES_FILE}")
  target_link_libraries(notices_smoke PRIVATE Isoduo::Desktop)
  add_test(NAME notices_smoke COMMAND notices_smoke)
endif()
