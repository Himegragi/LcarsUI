# Stand-alone consumer: sees LcarsUI only through its public targets and include/ headers.
cmake_minimum_required(VERSION 3.25)
project(LcarsConsumer LANGUAGES CXX)
set(LCARS_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../.." CACHE PATH "LcarsUI source tree")
set(LCARS_BUILD_DESKTOP OFF CACHE BOOL "" FORCE)
set(LCARS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${LCARS_SOURCE_DIR}" lcars EXCLUDE_FROM_ALL)
add_executable(profile_alias main.cpp)
set_target_properties(profile_alias PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(profile_alias PRIVATE /utf-8)
endif()
target_link_libraries(profile_alias PRIVATE LcarsUI::Core LcarsUI::AppearanceMaterial)
enable_testing()
add_test(NAME profile_alias COMMAND profile_alias)
