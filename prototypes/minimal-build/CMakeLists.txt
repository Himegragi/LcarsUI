cmake_minimum_required(VERSION 3.25)
project(LcarsBuildPrototype LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This throwaway probe targets native Windows")
endif()
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
option(PROBE_BUILD_OPENGL "Build the Win32/OpenGL probe" ON)
if(MSVC)
  add_compile_options(/utf-8)
endif()

# Offline, pinned source headers. A changed file is an explicit configure error.
if(PROBE_BUILD_OPENGL)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/third_party/glcorearb.h" gl_hash)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/third_party/KHR/khrplatform.h" khr_hash)
if(NOT gl_hash STREQUAL "65fae555a8b3b5709099837001e0c14228b3e45651e2d0fa2ad22649833921e1")
  message(FATAL_ERROR "glcorearb.h checksum mismatch")
endif()
if(NOT khr_hash STREQUAL "7b1e01aaa7ad8f6fc34b5c7bdf79ebf5189bb09e2c4d2e79fc5d350623d11e83")
  message(FATAL_ERROR "khrplatform.h checksum mismatch")
endif()
endif()

add_library(probe_core STATIC core.cpp)
target_include_directories(probe_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(core_consumer core_consumer.cpp)
target_link_libraries(core_consumer PRIVATE probe_core)
if(PROBE_BUILD_OPENGL)
  add_executable(gl_probe gl_probe.cpp)
  target_include_directories(gl_probe PRIVATE third_party)
  target_compile_definitions(gl_probe PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_link_libraries(gl_probe PRIVATE probe_core opengl32 user32 gdi32)
endif()
