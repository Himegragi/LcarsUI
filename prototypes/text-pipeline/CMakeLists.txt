function(unpack name folder expected)
  set(archive "${CMAKE_CURRENT_SOURCE_DIR}/third_party/${name}.tar.gz")
  file(SHA256 "${archive}" actual)
  if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "${name} source checksum mismatch")
  endif()
  file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/sources")
  if(NOT EXISTS "${CMAKE_BINARY_DIR}/sources/${name}.ready")
    # Upstream CLAUDE.md is a non-build symlink; Windows tar cannot create it here.
    execute_process(COMMAND tar -xf "${archive}" -C "${CMAKE_BINARY_DIR}/sources" --exclude "*/CLAUDE.md" COMMAND_ERROR_IS_FATAL ANY)
    file(WRITE "${CMAKE_BINARY_DIR}/sources/${name}.ready" "${actual}")
  endif()
  set(${name}_SOURCE "${CMAKE_BINARY_DIR}/sources/${folder}" PARENT_SCOPE)
endfunction()

# Narrow compatibility patch: this header helper is unused in one upstream TU;
# HarfBuzz's own -Wunused pragma promotes that warning to an error on clang-cl.
# Keep all diagnostic checks enabled; mark only the internal header helper inline.
set(hb_header "${harfbuzz_SOURCE}/src/hb-ot-layout-gsubgpos.hh")
file(READ "${hb_header}" hb_original)
string(REPLACE "\nstatic bool\ncontext_depend_sets_are_cached" "\nstatic inline bool\ncontext_depend_sets_are_cached" hb_patched "${hb_original}")
if(NOT hb_original STREQUAL hb_patched)
  file(WRITE "${hb_header}" "${hb_patched}")
endif()
add_subdirectory("${harfbuzz_SOURCE}" harfbuzz EXCLUDE_FROM_ALL)
