cmake_minimum_required(VERSION 3.25)
project(LcarsStatePrototype LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_library(state_model STATIC state_model.cpp)
target_include_directories(state_model PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(state_probe native_probe.cpp)
target_compile_definitions(state_probe PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(state_probe PRIVATE /utf-8)
target_link_libraries(state_probe PRIVATE state_model user32 gdi32)
