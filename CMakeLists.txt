cmake_minimum_required(VERSION 3.25)
project(LcarsUI VERSION 0.1.0 LANGUAGES C CXX)
option(LCARS_BUILD_DESKTOP "Build the Windows/OpenGL desktop host" ON)
option(LCARS_BUILD_TESTS "Build behavior tests" ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
function(lcars_target target)
  # C++20 is the framework floor: required, never silently decayed, no vendor extensions.
  set_target_properties(${target} PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
  target_compile_features(${target} PUBLIC cxx_std_20)
  target_include_directories(${target} PUBLIC "${PROJECT_SOURCE_DIR}/include")
  if(MSVC)
    target_compile_options(${target} PRIVATE /utf-8 /W4)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endfunction()
add_library(lcars_core STATIC src/core.cpp src/layout.cpp)
lcars_target(lcars_core)
add_library(LcarsUI::Core ALIAS lcars_core)
add_library(lcars_appearance_web STATIC src/appearance_web.cpp)

add_library(lcars_appearance_material STATIC src/appearance_material.cpp)
lcars_target(lcars_appearance_material)

  add_executable(lcars_tests tests/tests.cpp)
  lcars_target(lcars_tests)
  target_include_directories(lcars_tests PRIVATE src)
  target_link_libraries(lcars_tests PRIVATE lcars_core lcars_appearance_web lcars_appearance_material)
  add_test(NAME core_regressions COMMAND lcars_tests)
  foreach(suite components layout edit_form)
    add_executable(lcars_${suite}_tests tests/${suite}.cpp)

    target_include_directories(lcars_${suite}_tests PRIVATE src)

    add_test(NAME ${suite} COMMAND lcars_${suite}_tests)
  endforeach()
  target_link_libraries(lcars_edit_form_tests PRIVATE lcars_edit_form)
  # AC02: configure, build and run a separate project that only sees public targets/headers.
  add_test(NAME consumer_example COMMAND "${CMAKE_CTEST_COMMAND}"
    --build-and-test "${PROJECT_SOURCE_DIR}/examples/consumer" "${CMAKE_BINARY_DIR}/consumer"
    --build-generator "${CMAKE_GENERATOR}"
    --build-options "-DLCARS_SOURCE_DIR=${PROJECT_SOURCE_DIR}" "-DCMAKE_BUILD_TYPE=${CMAKE_BUILD_TYPE}"
