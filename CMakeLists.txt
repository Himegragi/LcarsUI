cmake_minimum_required(VERSION 3.25)
project(LcarsUI VERSION 0.1.0 LANGUAGES C CXX)
option(LCARS_BUILD_DESKTOP "Build the Windows/OpenGL settings demo" ON)
option(LCARS_BUILD_TESTS "Build behavior and persistence tests" ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
function(lcars_target target)
  # C++20 is the framework floor: required, never silently decayed, no vendor extensions.
  set_target_properties(${target} PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
  target_compile_features(${target} PUBLIC cxx_std_20)
  target_include_directories(${target} PUBLIC "${PROJECT_SOURCE_DIR}/include")
  if(MSVC)
    target_compile_options(${target} PRIVATE /utf-8 /W4)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endfunction()
add_library(lcars_core STATIC src/core.cpp src/layout.cpp)
lcars_target(lcars_core)
add_library(LcarsUI::Core ALIAS lcars_core)
add_library(lcars_appearance_web STATIC src/appearance_web.cpp)
lcars_target(lcars_appearance_web)
add_library(LcarsUI::AppearanceWeb ALIAS lcars_appearance_web)
add_library(lcars_appearance_material STATIC src/appearance_material.cpp)
lcars_target(lcars_appearance_material)
add_library(LcarsUI::AppearanceMaterial ALIAS lcars_appearance_material)
add_library(lcars_settings STATIC examples/settings/settings.cpp)
lcars_target(lcars_settings)
target_include_directories(lcars_settings PUBLIC "${PROJECT_SOURCE_DIR}/examples/settings")
target_link_libraries(lcars_settings PUBLIC lcars_core)
# V01 minimal edit form: application code assembled only from public headers.
add_library(lcars_edit_form STATIC examples/edit_form/edit_form.cpp)
lcars_target(lcars_edit_form)
target_include_directories(lcars_edit_form PUBLIC "${PROJECT_SOURCE_DIR}/examples/edit_form")
target_link_libraries(lcars_edit_form PUBLIC lcars_core)
if(LCARS_BUILD_TESTS)
  enable_testing()
  add_executable(lcars_tests tests/tests.cpp)
  lcars_target(lcars_tests)
  target_link_libraries(lcars_tests PRIVATE lcars_settings lcars_appearance_web lcars_appearance_material)
  add_test(NAME behavior_and_persistence COMMAND lcars_tests)
  foreach(suite components layout edit_form)
    add_executable(lcars_${suite}_tests tests/${suite}.cpp)
    lcars_target(lcars_${suite}_tests)
    target_link_libraries(lcars_${suite}_tests PRIVATE lcars_core lcars_appearance_web lcars_appearance_material)
    add_test(NAME ${suite} COMMAND lcars_${suite}_tests)
  endforeach()
  target_link_libraries(lcars_edit_form_tests PRIVATE lcars_edit_form)
  # AC02: configure, build and run a separate project that only sees public targets/headers.
  add_test(NAME consumer_example COMMAND "${CMAKE_CTEST_COMMAND}"
    --build-and-test "${PROJECT_SOURCE_DIR}/examples/consumer" "${CMAKE_BINARY_DIR}/consumer"
    --build-generator "${CMAKE_GENERATOR}"
    --build-options "-DLCARS_SOURCE_DIR=${PROJECT_SOURCE_DIR}" "-DCMAKE_BUILD_TYPE=${CMAKE_BUILD_TYPE}"
      "-DCMAKE_CXX_COMPILER=${CMAKE_CXX_COMPILER}" "-DCMAKE_C_COMPILER=${CMAKE_C_COMPILER}"
      "-DCMAKE_MAKE_PROGRAM=${CMAKE_MAKE_PROGRAM}"
    --test-command "${CMAKE_CTEST_COMMAND}" --output-on-failure)
endif()
if(LCARS_BUILD_DESKTOP)
  if(NOT WIN32)
    message(FATAL_ERROR "The MVP desktop host requires Windows. Use -DLCARS_BUILD_DESKTOP=OFF for core tests.")
  endif()
  include(cmake/TextDependencies.cmake)
  add_library(lcars_desktop STATIC src/text.cpp src/renderer_opengl.cpp)
  lcars_target(lcars_desktop)
  target_include_directories(lcars_desktop PRIVATE third_party)
  target_compile_definitions(lcars_desktop PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE _WIN32_WINNT=0x0A00)
  target_link_libraries(lcars_desktop PUBLIC lcars_core PRIVATE harfbuzz freetype opengl32 user32 gdi32 imm32 shell32)
  add_library(lcars_settings_host STATIC examples/settings/desktop.cpp)
  lcars_target(lcars_settings_host)
  target_include_directories(lcars_settings_host PRIVATE src)
  target_compile_definitions(lcars_settings_host PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE _WIN32_WINNT=0x0A00)
  target_link_libraries(lcars_settings_host PUBLIC lcars_desktop)
  foreach(skin web material)
    add_executable(lcars_settings_${skin} examples/settings/main.cpp)
    lcars_target(lcars_settings_${skin})
    target_compile_definitions(lcars_settings_${skin} PRIVATE LCARS_APPEARANCE_${skin})
    target_link_libraries(lcars_settings_${skin} PRIVATE lcars_settings lcars_settings_host lcars_appearance_${skin})
  endforeach()
endif()
